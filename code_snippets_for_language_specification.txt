# class definitions
# a Cclass inherites from Object implictly
# method overloading based on signature
# real private class members, no name mangling
# access modifiers private, protected (so acces to own class and subclass only), module (acces only in the file, like pyhton does) public
# file example.pys
class Car{
	# attribute declaration
	private string brand;
	# constructor, same name as class, not return type, no self is used
	public Car(string brand){
		# self reference own instance
		this.brand = brand;
	}
	
	public drive(){
		print("Moving!");
	}
	
	module string brand(){
		return this.brand;
	}
}

# inheritance
class Truck inherits Car{
	# attribute declaration
	# constructor, same name as class, not return type, no self is used
	public Truck(string brand){
		super(brand);
	}
	
	public drive(){
		super.drive()
		print("\n With 18 wheels!");
	}

}

# use
Car c1 = Car("Toyota");
Truck t1 = Truck("DAF");

c1.drive();
t1.drive(); 
print("brand of the truck is {t1.brand()");
